# FedCondGraphRAG — core method environment (preprocess / fl-train / train /
# infer). Reproducible on a fresh server in two commands:
#
#   conda create -n fedrag python=3.11 -y && conda activate fedrag
#   pip install -r requirements.txt
#   python -m spacy download en_core_web_sm
#
# torch/torch_scatter are pinned to 2.9.1+cu128 via the extra index/find-links
# below. Needs to be at least torch>=2.7.0+cu128 for Blackwell (RTX 50-series,
# CUDA capability sm_120) GPU support -- an older pin (2.5.0+cu124, this
# project's original dev machine had an Ada/RTX 4060) fails on sm_120 with
# "CUDA capability sm_120 is not compatible with the current PyTorch
# installation". cu128 wheels remain compatible with older architectures too
# (Ada/sm_89 and earlier), so this one pin covers both. Confirmed a prebuilt
# torch_scatter wheel exists for this exact combo at data.pyg.org/whl before
# picking it -- pyg's wheel builds lag new torch releases, not every
# torch+cuda combo has one.
#
# This file is only for the core method. The baselines under
# fedcond_grag/baselines/ (HippoRAG, GRAG, FLARE, ComoRAG, G-Retriever) each
# need extra, heavier, sometimes fragile dependencies for a fair-comparison
# run -- see requirements-baselines.txt, kept separate so installing it
# never risks the core method's own working environment.
